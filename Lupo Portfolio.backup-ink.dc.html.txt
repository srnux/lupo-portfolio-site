<script type="text/x-dc" data-dc-script data-props="{&quot;email&quot;:{&quot;editor&quot;:&quot;text&quot;,&quot;default&quot;:&quot;lupo@lupoart.de&quot;,&quot;tsType&quot;:&quot;string&quot;,&quot;section&quot;:&quot;Content&quot;},&quot;portfolioFile&quot;:{&quot;editor&quot;:&quot;text&quot;,&quot;default&quot;:&quot;assets/lupo-portfolio.png&quot;,&quot;tsType&quot;:&quot;string&quot;,&quot;section&quot;:&quot;Content&quot;},&quot;illustrationCount&quot;:{&quot;editor&quot;:&quot;int&quot;,&quot;default&quot;:9,&quot;min&quot;:3,&quot;max&quot;:24,&quot;tsType&quot;:&quot;number&quot;,&quot;section&quot;:&quot;Gallery&quot;},&quot;conceptCount&quot;:{&quot;editor&quot;:&quot;int&quot;,&quot;default&quot;:6,&quot;min&quot;:2,&quot;max&quot;:24,&quot;tsType&quot;:&quot;number&quot;,&quot;section&quot;:&quot;Gallery&quot;}}">
const COPY = {
  en: {
    navIllu: "Illustration", navConcept: "Concept Art", navAbout: "About me",
    aboutTitle: "About me", download: "Portfolio", contact: "Contact",
    more: "more", less: "less",
    aboutBody: "I'm Lupo, a design and illustration student working between hand-drawn character work and moody concept art for games and books. I draw in ink and paint digitally, and I like faces that look like they have been through something. Currently taking on commissions, collaborations and internships."
  },
  de: {
    navIllu: "Illustration", navConcept: "Concept Art", navAbout: "Über mich",
    aboutTitle: "Über mich", download: "Portfolio", contact: "Kontakt",
    more: "mehr", less: "weniger",
    aboutBody: "Ich bin Lupo, Student für Design und Illustration – zwischen handgezeichneter Charakterarbeit und stimmungsvollem Concept Art für Spiele und Bücher. Ich zeichne mit Tusche und male digital, und ich mag Gesichter, die etwas erlebt haben. Für Aufträge, Kollaborationen und Praktika bin ich offen."
  }
};

class Component extends DCLogic {
  state = { lang: "en", menuOpen: false, barHidden: false, nIllu: 3, nConcept: 2, contactOpen: false };

  constructor(props) {
    super(props);
    this.scroller = React.createRef();
    this.refs1 = React.createRef();
    this.refs2 = React.createRef();
    this.refs3 = React.createRef();
    this.refs4 = React.createRef();
    this.lastY = 0;
    this.onScroll = this.onScroll.bind(this);
  }

  componentDidMount() {
    const el = this.scroller.current;
    if (el) el.addEventListener("scroll", this.onScroll, { passive: true });
  }
  componentWillUnmount() {
    const el = this.scroller.current;
    if (el) el.removeEventListener("scroll", this.onScroll);
  }
  onScroll() {
    const el = this.scroller.current;
    if (!el) return;
    const y = el.scrollTop;
    const hidden = y > 120 && y > this.lastY + 4;
    const shown = y < 60 || y < this.lastY - 4;
    if (hidden && !this.state.barHidden) this.setState({ barHidden: true, menuOpen: false });
    else if (shown && this.state.barHidden) this.setState({ barHidden: false });
    this.lastY = y;
  }
  goTo(ref) {
    const el = this.scroller.current, target = ref.current;
    if (el && target) el.scrollTo({ top: target.offsetTop, behavior: "smooth" });
    this.setState({ menuOpen: false });
  }

  pics(prefix, total, shown, label) {
    const out = [];
    for (let i = 0; i < Math.min(shown, total); i++) out.push({ id: prefix + "-" + (i + 1), label: label + " " + (i + 1) });
    return out;
  }

  renderVals() {
    const t = COPY[this.state.lang];
    const totalIllu = this.props.illustrationCount ?? 9;
    const totalConcept = this.props.conceptCount ?? 6;
    const email = this.props.email ?? "lupo@lupoart.de";
    return {
      t, email,
      mailto: "mailto:" + email,
      portfolioHref: this.props.portfolioFile ?? "assets/lupo-portfolio.png",
      scrollerRef: this.scroller,
      s1: this.refs1, s2: this.refs2, s3: this.refs3, s4: this.refs4,
      barT: this.state.barHidden ? "translateY(-115%)" : "translateY(0)",
      menuOpen: this.state.menuOpen,
      contactOpen: this.state.contactOpen,
      enColor: this.state.lang === "en" ? "#e2685c" : "rgba(242,236,230,0.75)",
      deColor: this.state.lang === "de" ? "#e2685c" : "rgba(242,236,230,0.75)",
      illu: this.pics("illu", totalIllu, this.state.nIllu, "A3"),
      concept: this.pics("concept", totalConcept, this.state.nConcept, "Concept"),
      illuLabel: this.state.nIllu >= totalIllu ? t.less : t.more,
      conceptLabel: this.state.nConcept >= totalConcept ? t.less : t.more,
      setEn: () => this.setState({ lang: "en" }),
      setDe: () => this.setState({ lang: "de" }),
      toggleMenu: () => this.setState(s => ({ menuOpen: !s.menuOpen })),
      toggleContact: () => this.setState(s => ({ contactOpen: !s.contactOpen })),
      moreIllu: () => this.setState(s => ({ nIllu: s.nIllu >= totalIllu ? 3 : totalIllu })),
      moreConcept: () => this.setState(s => ({ nConcept: s.nConcept >= totalConcept ? 2 : totalConcept })),
      goHome: () => this.goTo(this.refs1),
      goIllu: () => this.goTo(this.refs2),
      goConcept: () => this.goTo(this.refs3),
      goAbout: () => this.goTo(this.refs4)
    };
  }
}
</script>
